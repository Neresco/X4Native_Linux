cmake_minimum_required(VERSION 3.20)
project(x4mp_join LANGUAGES CXX)
add_library(x4mp_join SHARED join_bridge.cpp)
target_compile_features(x4mp_join PRIVATE cxx_std_17)
target_include_directories(x4mp_join PRIVATE ../../sdk)
set_target_properties(x4mp_join PROPERTIES PREFIX "" OUTPUT_NAME "x4mp_join")
if(MSVC)
    target_compile_definitions(x4mp_join PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(join_bridge_test tests/bridge.cpp)
    target_compile_features(join_bridge_test PRIVATE cxx_std_17)
    target_include_directories(join_bridge_test PRIVATE ../../sdk)
    if(MSVC)
        target_compile_definitions(join_bridge_test PRIVATE _CRT_SECURE_NO_WARNINGS)
    endif()
    add_test(NAME join_bridge COMMAND join_bridge_test)
endif()
